cmake_minimum_required(VERSION 3.8)
project(controller_tracking_probe)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(measured_controller SHARED src/measured_controller.cpp)
target_compile_features(measured_controller PUBLIC cxx_std_17)
ament_target_dependencies(measured_controller nav2_core pluginlib rclcpp rclcpp_lifecycle nav_msgs std_msgs geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS measured_controller LIBRARY DESTINATION lib)
ament_package()
